// API Documentation

Backend REST API — /api/v1

The backend exposes an OpenAPI / Swagger schema at:

  • Swagger UI – /docs
  • OpenAPI JSON – /openapi.json
The Documentation page embeds the live Swagger UI so you can browse and try requests without leaving the application.

Authentication

All /api/v1/* endpoints require a valid access token. The frontend signs in via /api/v1/auth/login, which sets an httpOnly cookie sent automatically on subsequent requests together with a double-submit CSRF token (X-CSRF-Token header).

Versioning

The API is mounted under /api/v1. Breaking changes will be introduced behind a new version prefix; additive changes (new endpoints, optional fields) are released in-place.

Step-by-Step: Using Swagger UI

Open Swagger UI

Navigate to http://localhost:8000/docs

Locate the Login Endpoint

Under the Auth section, click /api/v1/auth/login.

Click "Try it out"

Enable the interactive request form by clicking the "Try it out" button.

Enter Login Details

Fill in the Request body:

{
  "email": "<your-intel-email>",
  "password": "<your-password>"
}
Note: Passwords are set up by the admin. Contact the system admin for your credentials. After SSO is completed, your Intel password will work directly.

Execute and Copy Token

Click "Execute". In the Responses section, find the 200 response and copy the access_token value.

Authorize in Swagger

Scroll to the top, click "Authorize", paste your token, and confirm. The lock icon will close — you're now authenticated. Note: some endpoints require admin role.

Explore Endpoints

You can now try any available API endpoint from the Swagger UI.

Using the API Programmatically

Select your language below to see a working example. Make sure you have your credentials ready, or obtain the token programmatically as shown.

import requests

# Step 1: Login to get access_token
login_url = "http://localhost:8000/api/v1/auth/login"
login_data = {"email": "your@email.com", "password": "yourpassword"}
resp = requests.post(login_url, json=login_data)
access_token = resp.json()["access_token"]

# Step 2: Use access_token in Authorization header
headers = {"Authorization": f"Bearer {access_token}"}
api_url = "http://localhost:8000/api/v1/some_endpoint"
response = requests.get(api_url, headers=headers)
print(response.json())
// JavaScript (Node.js)
const fetch = require('node-fetch');

(async () => {
  // Step 1: Login
  const loginResp = await fetch('http://localhost:8000/api/v1/auth/login', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ email: 'your@email.com', password: 'yourpassword' })
  });
  const { access_token } = await loginResp.json();

  // Step 2: Call a protected endpoint
  const apiResp = await fetch('http://localhost:8000/api/v1/some_endpoint', {
    headers: { 'Authorization': `Bearer ${access_token}` }
  });
  console.log(await apiResp.json());
})();

import java.net.*;
import java.io.*;

public class ApiExample {
  public static void main(String[] args) throws Exception {
    // Step 1: Login to get access_token
    URL loginUrl = new URL("http://localhost:8000/api/v1/auth/login");
    HttpURLConnection conn = (HttpURLConnection) loginUrl.openConnection();
    conn.setRequestMethod("POST");
    conn.setRequestProperty("Content-Type", "application/json");
    conn.setDoOutput(true);
    String body = "{\"email\":\"your@email.com\",\"password\":\"yourpassword\"}";
    try (OutputStream os = conn.getOutputStream()) {
      os.write(body.getBytes());
    }
    BufferedReader br = new BufferedReader(new InputStreamReader(conn.getInputStream()));
    StringBuilder sb = new StringBuilder();
    String line;
    while ((line = br.readLine()) != null) sb.append(line);
    String accessToken = sb.toString().split("\"access_token\":\"")[1].split("\"")[0];

    // Step 2: Use access_token
    URL apiUrl = new URL("http://localhost:8000/api/v1/some_endpoint");
    HttpURLConnection apiConn = (HttpURLConnection) apiUrl.openConnection();
    apiConn.setRequestProperty("Authorization", "Bearer " + accessToken);
    BufferedReader apiBr = new BufferedReader(new InputStreamReader(apiConn.getInputStream()));
    while ((line = apiBr.readLine()) != null) System.out.println(line);
  }
}
using System;
using System.Net.Http;
using System.Net.Http.Headers;
using System.Text;
using System.Threading.Tasks;

class Program {
  static async Task Main() {
    var client = new HttpClient();

    // Step 1: Login
    var loginData = new StringContent(
      "{\"email\":\"your@email.com\",\"password\":\"yourpassword\"}",
      Encoding.UTF8, "application/json"
    );
    var loginResp = await client.PostAsync("http://localhost:8000/api/v1/auth/login", loginData);
    var loginJson = await loginResp.Content.ReadAsStringAsync();
    var accessToken = loginJson.Split("\"access_token\":\"")[1].Split('"')[0];

    // Step 2: Call a protected endpoint
    client.DefaultRequestHeaders.Authorization =
      new AuthenticationHeaderValue("Bearer", accessToken);
    var apiResp = await client.GetAsync("http://localhost:8000/api/v1/some_endpoint");
    Console.WriteLine(await apiResp.Content.ReadAsStringAsync());
  }
}
// C++ — requires libcurl
#include <curl/curl.h>
#include <string>
#include <iostream>

int main() {
  CURL *curl = curl_easy_init();
  if (!curl) return 1;

  struct curl_slist *headers = nullptr;
  headers = curl_slist_append(headers, "Content-Type: application/json");

  // Step 1: Login
  const char *loginBody =
    "{\"email\":\"your@email.com\",\"password\":\"yourpassword\"}";
  curl_easy_setopt(curl, CURLOPT_URL, "http://localhost:8000/api/v1/auth/login");
  curl_easy_setopt(curl, CURLOPT_HTTPHEADER, headers);
  curl_easy_setopt(curl, CURLOPT_POSTFIELDS, loginBody);
  // ... capture response, parse access_token ...

  // Step 2: Use access_token
  std::string authHeader = "Authorization: Bearer " + access_token;
  struct curl_slist *apiHeaders = nullptr;
  apiHeaders = curl_slist_append(apiHeaders, authHeader.c_str());
  curl_easy_setopt(curl, CURLOPT_URL, "http://localhost:8000/api/v1/some_endpoint");
  curl_easy_setopt(curl, CURLOPT_HTTPGET, 1L);
  curl_easy_setopt(curl, CURLOPT_HTTPHEADER, apiHeaders);
  curl_easy_perform(curl);

  curl_easy_cleanup(curl);
  return 0;
}
